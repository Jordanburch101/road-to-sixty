# Website Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The public site in `apps/web` (Astro, static, on Vercel): the home page as mocked, a download page per platform with the unsigned-app steps, Privacy, Terms and a not-found page.

**Architecture:** Astro builds plain HTML at build time. The shared React components from `packages/ui` (`Fog`, `Button`) are rendered to HTML there, and no React reaches the browser. The one script picks the download for the visitor's computer, and its rule is a tested plain function. The fog gains roads in their own colours and timing, for the "one road" band. The class icons move to `packages/ui` so the site and the app share them.

**Tech Stack:** Astro 7.3.5, @astrojs/react 7.0.0, Tailwind 4.3.3 (`@tailwindcss/vite`), React 19.3.0, Bun 1.4.2 tests, TypeScript 7.0.2 (`tsc`).

**Spec:** `docs/superpowers/specs/2026-10-15-website-design.md`. Mockups: the "Website" row of https://claude.ai/artifact/M1btBtHDuiExpSuHU7Pg8Z (`Site-Home` top, `Site-Below-2` the rest, `Site-Phone`, `Site-Privacy`).

## Decisions made while planning

- **`astro check` refuses TypeScript 7** (tried 2026-10-15). `typecheck` is `tsc` over `src/**/*.ts(x)` with Astro's client types, and `astro build` catches mistakes in `.astro` templates.
- **Without the script, the main button reads "Download for Mac or Windows"** and goes to `/download`, which offers both. The script turns it into the visitor's platform. Phones, tablets and Linux keep the both-platforms button.
- **The download page starts the file with a one-second `meta refresh`,** so it works without JavaScript, and it also has a plain link.
- **Two fogs on one page each keep their own speed.** The hero runs at 22 s and the band at 9 s, so `Fog` puts each dot's loop length inline, beside its delay.
- **The site's app picture hardcodes its four made-up characters' class colours,** so `format.ts`'s colour list stays in the app. Only the icons move.
- **The install-step screenshots are added when Jordan supplies them** (spec, open items). The download pages ship with the steps in words.

`packages/ui/src/Fog.tsx`, whole file:

```tsx
import { useMemo } from "react";
import { fogAnimation, fogDelay, fogDots, type Point, type Timing } from "./dots";

// Draws the fog grid at a fixed size; the parent clips and fades it.
// animate is the loop's length in seconds; without it the fog is still.
// The length goes on each dot too, so two fogs on a page keep their speeds.
export function Fog({
  width,
  height,
  step,
  roads,
  marks = [],
  animate,
  colours,
  timing,
}: {
  width: number;
  height: number;
  step: number;
  roads: Point[][];
  marks?: Point[];
  animate?: number;
  colours?: string[];
  timing?: Timing[];
}) {
  const { lit, edge, fog } = useMemo(() => fogDots(width, height, step, roads, timing), [width, height, step, roads, timing]);
  const moving = (at: number) =>
    animate ? { className: "rts-fog-road", style: { animationDelay: fogDelay(at, animate), animationDuration: `${animate}s` } } : { style: {} };
  return (
    <svg aria-hidden="true" width={width} height={height} viewBox={`0 0 ${width} ${height}`} className="pointer-events-none absolute left-0 top-0">
      {animate ? <style>{fogAnimation(animate)}</style> : null}
      {fog.map(([x, y]) => <circle key={`f${x},${y}`} cx={x} cy={y} r={1.1} fill="rgba(154,149,168,0.16)" />)}
      {edge.map(([x, y, at]) => <circle key={`e${x},${y}`} cx={x} cy={y} r={1.4} fill="rgba(154,149,168,0.4)" {...moving(at)} />)}
      {lit.map(([x, y, at, road]) => {
        const m = moving(at);
        const colour = colours?.[road];
        return (
          <circle
            key={`l${x},${y}`}
            cx={x}
            cy={y}
            r={1.8}
            fillOpacity={0.85}
            className={`${colour ? "" : "fill-accent"} ${m.className ?? ""}`}
            style={{ ...m.style, ...(colour ? { fill: colour } : {}) }}
          />
        );
      })}
      {marks.map(([x, y]) => (
        <g key={`m${x},${y}`}>
          <circle cx={x} cy={y} r={8} fill="none" className="stroke-accent" strokeOpacity={0.35} />
          <circle cx={x} cy={y} r={3} className="fill-accent" />
        </g>
      ))}
    </svg>
  );
}
```

(`{animate ? … : null}` also fixes the "0" that `animate={0}` used to render, the plan-1 minor.)

// Build settings, from Vercel's environment (or .env locally). The build
// stops when one is missing, rather than shipping an empty link.
export function need(name: string, value: string | undefined): string {
  if (!value) throw new Error(`${name} isn't set: see apps/web/.env.example`);

  return `${DOWNLOADS}/app/${platform === "mac" ? "Road-to-Sixty.dmg" : "Road-to-Sixty-Setup.exe"}`;

`apps/web/src/download.ts`:

```ts
// Which download to offer: the visitor's own computer when it is a Mac or
// Windows, both otherwise. iPads report a Mac browser, so a touch screen
// counts against it.
export type Choice = "mac" | "windows" | "choose";

export function pick(nav: { userAgent: string; platform?: string; maxTouchPoints?: number }): Choice {
  const ua = nav.userAgent;
  const platform = nav.platform ?? "";
  if (/iPhone|iPad|iPod|Android/i.test(ua)) return "choose";
  if (/Windows/i.test(ua) || /^Win/i.test(platform)) return "windows";
  if (/Macintosh/i.test(ua) || /^Mac/i.test(platform)) return (nav.maxTouchPoints ?? 0) > 1 ? "choose" : "mac";
  return "choose";
}

export const LABEL: Record<Choice, string> = {
  mac: "Download for macOS",
  windows: "Download for Windows",
  choose: "Download for Mac or Windows",
};

export const PAGE: Record<Choice, string> = { mac: "/download/mac", windows: "/download/windows", choose: "/download" };

// Points every [data-download] button at this computer's download, and the
// "also for" line at the other one. Without this, the page offers both.
export function wireDownloads() {
  const data = (navigator as Navigator & { userAgentData?: { platform?: string } }).userAgentData;
  const choice = pick({ userAgent: navigator.userAgent, platform: data?.platform ?? navigator.platform, maxTouchPoints: navigator.maxTouchPoints });
  for (const a of document.querySelectorAll<HTMLAnchorElement>("[data-download]")) {
    a.href = PAGE[choice];
    const label = a.querySelector<HTMLElement>("[data-label]");
    if (label) label.textContent = LABEL[choice];
    a.querySelector<HTMLElement>("[data-apple]")?.toggleAttribute("hidden", choice !== "mac");
  }
  for (const line of document.querySelectorAll<HTMLElement>("[data-also]")) {
    if (choice === "choose") continue;
    const other = choice === "mac" ? "windows" : "mac";
    const link = line.querySelector<HTMLAnchorElement>("a")!;
    link.href = PAGE[other];
    link.textContent = other === "mac" ? "Mac" : "Windows";
    line.hidden = false;
  }
}
```

import type { Point, Timing } from "@road-to-sixty/ui/dots";

// The hero's two wandering roads, the same every build: a seeded random
// walk that turns back towards the middle near the edges.
function seeded(seed: number) {
  let s = seed | 0;
  return () => {
    s = (s + 0x6d2b79f5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function walk(rnd: () => number, width: number, height: number, x: number, y: number, a: number, steps: number, len: number): Point[] {
  const pts: Point[] = [[x, y]];
  for (let i = 0; i < steps; i++) {
    a += (rnd() - 0.5) * 0.9;
    if (x < 30 || x > width - 30 || y < 30 || y > height - 30) {
      const back = Math.atan2(height / 2 - y, width / 2 - x);
      a += Math.atan2(Math.sin(back - a), Math.cos(back - a)) * 0.3;
    }
    x += Math.cos(a) * len;
    y += Math.sin(a) * len;
    pts.push([x, y]);
  }
  return pts;
}

export const HERO = { width: 1440, height: 900 };
const rnd = seeded(11);
export const HERO_ROADS: Point[][] = [
  walk(rnd, HERO.width, HERO.height, -20, 640, -0.4, 60, 30),
  walk(rnd, HERO.width, HERO.height, HERO.width + 20, 120, Math.PI + 0.3, 45, 30),
];
export const HERO_MARKS: Point[] = HERO_ROADS.flatMap((road) => road.filter((_, i) => i >= 7 && (i - 7) % 10 === 0));

// The "one road" band: Desk's road (blue) and Laptop's (cream) light up

];
